# RAG Service + Eval Harness — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a RAG service (ingest → hybrid retrieve → answer with citations) plus a rigorous eval harness (retrieval metrics + LLM-as-judge answer quality, cost/latency, CI regression gate) and a Next.js dashboard.

**Architecture:** Clean architecture, dependencies pointing inward. A pure `domain` (entities + Protocol ports), pure `metrics`/`chunking` modules, and infra `adapters` (fastembed, pgvector, Claude) wired by `services`. The LLM appears in exactly two adapter files; SQL only in the persistence adapters. Local embeddings keep the core key-free; Claude is gated behind `ANTHROPIC_API_KEY`.

**Tech Stack:** Python 3.12 (uv) · FastAPI · Typer · psycopg3 · Postgres + pgvector · fastembed (bge-small, ONNX) · anthropic SDK (Opus 4.8) · pydantic v2 · pytest · mypy strict · ruff · Next.js (TS strict) · Docker Compose · GitHub Actions.

from __future__ import annotations
import math
from collections.abc import Sequence


def recall_at_k(ranked_ids: Sequence[str], relevant_ids: set[str], k: int) -> float:
    if not relevant_ids:
        return 0.0
    top = set(ranked_ids[:k])
    return len(top & relevant_ids) / len(relevant_ids)


def precision_at_k(ranked_ids: Sequence[str], relevant_ids: set[str], k: int) -> float:
    if k <= 0:
        return 0.0
    top = ranked_ids[:k]
    hits = sum(1 for cid in top if cid in relevant_ids)
    return hits / k


def mrr(ranked_ids: Sequence[str], relevant_ids: set[str]) -> float:
    for i, cid in enumerate(ranked_ids):
        if cid in relevant_ids:
            return 1.0 / (i + 1)
    return 0.0


def ndcg_at_k(ranked_ids: Sequence[str], relevant_ids: set[str], k: int) -> float:
    if not relevant_ids:
        return 0.0
    dcg = sum(
        1.0 / math.log2(i + 2)
        for i, cid in enumerate(ranked_ids[:k])
        if cid in relevant_ids
    )
    ideal_hits = min(len(relevant_ids), k)
    idcg = sum(1.0 / math.log2(i + 2) for i in range(ideal_hits))
    return dcg / idcg if idcg > 0 else 0.0
```

from __future__ import annotations
from collections.abc import Sequence


def rrf_fuse(rankings: Sequence[Sequence[str]], k_const: int = 60) -> list[str]:
    """Reciprocal Rank Fusion: score = sum 1/(k_const + rank), rank 1-based."""
    scores: dict[str, float] = {}
    for ranking in rankings:
        for rank, item in enumerate(ranking, start=1):
            scores[item] = scores.get(item, 0.0) + 1.0 / (k_const + rank)
    return sorted(scores, key=lambda item: (-scores[item], item))
```

from __future__ import annotations
import re
from collections.abc import Iterable
from rag.domain.entities import Chunk

_WS = re.compile(r"\s+")


def _norm(text: str) -> str:
    return _WS.sub(" ", text.lower()).strip()


def relevant_chunk_ids(chunks: Iterable[Chunk], snippets: list[str]) -> set[str]:
    norm_snips = [_norm(s) for s in snippets if s.strip()]
    out: set[str] = set()
    for chunk in chunks:
        body = _norm(chunk.text)
        if any(s in body for s in norm_snips):
            out.add(chunk.id)
    return out

from __future__ import annotations
from rag.domain.entities import Chunk, Document


def chunk_document(doc: Document, target_words: int = 180, overlap_words: int = 30) -> list[Chunk]:
    words = doc.content.split()
    if not words:
        return []
    step = max(1, target_words - overlap_words)
    chunks: list[Chunk] = []
    ordinal = 0
    for start in range(0, len(words), step):
        window = words[start : start + target_words]
        text = " ".join(window)
        chunks.append(
            Chunk(
                id=f"{doc.id}:{ordinal}",
                document_id=doc.id,
                ordinal=ordinal,
                text=text,
                token_count=len(window),
            )
        )
        ordinal += 1
        if start + target_words >= len(words):
            break
    return chunks
```

from __future__ import annotations
from psycopg_pool import ConnectionPool
from rag.domain.entities import Chunk, Document, Vector

class PgChunkStore:
    def __init__(self, pool: ConnectionPool) -> None:
        self._pool = pool

    def add_document(self, doc: Document, chunks: list[tuple[Chunk, Vector]]) -> None:
        with self._pool.connection() as conn:
            conn.execute(
                "INSERT INTO documents (id, source, title, content) VALUES (%s,%s,%s,%s)"
                " ON CONFLICT (id) DO UPDATE SET content=EXCLUDED.content",
                (doc.id, doc.source, doc.title, doc.content),
            )
            for chunk, vec in chunks:
                conn.execute(
                    "INSERT INTO chunks (id, document_id, ordinal, text, embedding, token_count)"
                    " VALUES (%s,%s,%s,%s,%s,%s) ON CONFLICT (id) DO UPDATE SET"
                    " text=EXCLUDED.text, embedding=EXCLUDED.embedding",
                    (chunk.id, chunk.document_id, chunk.ordinal, chunk.text, vec, chunk.token_count),
                )
            conn.commit()

    def all_chunks(self) -> list[Chunk]:
        with self._pool.connection() as conn:
            rows = conn.execute(
                "SELECT id, document_id, ordinal, text, token_count FROM chunks"
            ).fetchall()
        return [Chunk(id=r[0], document_id=r[1], ordinal=r[2], text=r[3], token_count=r[4]) for r in rows]
```

from rag.domain.entities import Chunk, RetrievedChunk, Strategy
from rag.domain.ports import Embedder
from rag.metrics.fusion import rrf_fuse

class PgVectorRetriever:
    def __init__(self, pool: ConnectionPool, embedder: Embedder, pool_k: int = 50) -> None:
        self._pool = pool
        self._embedder = embedder
        self._pool_k = pool_k  # candidate depth per arm before fusion/cut

    def retrieve(self, query: str, k: int, strategy: Strategy) -> list[RetrievedChunk]:
        if strategy is Strategy.DENSE:
            ids = self._dense_ids(query, k)
        elif strategy is Strategy.SPARSE:
            ids = self._sparse_ids(query, k)
        else:  # HYBRID / HYBRID_RERANK
            dense = self._dense_ids(query, self._pool_k)
            sparse = self._sparse_ids(query, self._pool_k)
            ids = rrf_fuse([dense, sparse])[:k]
        chunks = self._load(ids)
        return [RetrievedChunk(chunk=c, score=1.0 / (i + 1), rank=i + 1)
                for i, c in enumerate(chunks)]

    def _dense_ids(self, query: str, k: int) -> list[str]:
        vec = self._embedder.embed([query])[0]
        with self._pool.connection() as conn:
            rows = conn.execute(
                "SELECT id FROM chunks ORDER BY embedding <=> %s LIMIT %s", (vec, k)
            ).fetchall()
        return [r[0] for r in rows]

    def _sparse_ids(self, query: str, k: int) -> list[str]:
        with self._pool.connection() as conn:
            rows = conn.execute(
                "SELECT id FROM chunks WHERE tsv @@ plainto_tsquery('english', %s)"
                " ORDER BY ts_rank_cd(tsv, plainto_tsquery('english', %s)) DESC LIMIT %s",
                (query, query, k),
            ).fetchall()
        return [r[0] for r in rows]

    def _load(self, ids: list[str]) -> list[Chunk]:
        if not ids:
            return []
        with self._pool.connection() as conn:
            rows = conn.execute(
                "SELECT id, document_id, ordinal, text, token_count FROM chunks WHERE id = ANY(%s)",
                (ids,),
            ).fetchall()
        by_id = {r[0]: Chunk(id=r[0], document_id=r[1], ordinal=r[2], text=r[3], token_count=r[4]) for r in rows}
        return [by_id[i] for i in ids if i in by_id]
```

from __future__ import annotations
from rag.chunking.chunker import chunk_document
from rag.domain.entities import Document
from rag.domain.ports import ChunkStore, Embedder

class IngestionService:
    def __init__(self, embedder: Embedder, store: ChunkStore) -> None:
        self._embedder = embedder
        self._store = store

    def ingest(self, docs: list[Document]) -> int:
        total = 0
        for doc in docs:
            chunks = chunk_document(doc)
            vectors = self._embedder.embed([c.text for c in chunks])
            self._store.add_document(doc, list(zip(chunks, vectors, strict=True)))
            total += len(chunks)
        return total
```
